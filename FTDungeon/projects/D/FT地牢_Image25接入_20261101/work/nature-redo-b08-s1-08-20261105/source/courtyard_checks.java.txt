    private static Vec3 courtFeet(JsonArray node, Fixture fixture) {
        double height = node.get(1).getAsDouble();
        BlockPos point = world(new BlockPos(node.get(0).getAsInt(), (int) Math.floor(height), node.get(2).getAsInt()), fixture);
        return new Vec3(point.getX() + 0.5, point.getY() + height - Math.floor(height), point.getZ() + 0.5);
    }

    private static AABB courtBody(Vec3 feet) {
        return new AABB(feet.x - 0.3, feet.y + 0.00001, feet.z - 0.3, feet.x + 0.3, feet.y + 1.8, feet.z + 0.3);
    }

    private static boolean courtStanding(ServerLevel level, Vec3 feet) {
        AABB support = new AABB(feet.x - 0.29, feet.y - 0.025, feet.z - 0.29, feet.x + 0.29, feet.y - 0.00001, feet.z + 0.29);
        return level.noCollision(courtBody(feet)) && !level.noCollision(support);
    }

    private static boolean courtEdge(ServerLevel level, Vec3 start, Vec3 end) {
        if (Math.abs(start.y - end.y) > 0.50001) return false;
        double raised = Math.max(start.y, end.y);
        Vec3 startUp = new Vec3(start.x, raised, start.z);
        Vec3 endUp = new Vec3(end.x, raised, end.z);
        return level.noCollision(courtBody(start).minmax(courtBody(startUp)))
                && level.noCollision(courtBody(startUp).minmax(courtBody(endUp)))
                && level.noCollision(courtBody(endUp).minmax(courtBody(end)));
    }

    private static void inspectCourt(Fixture fixture) {
        JsonArray route = fixture.spec.getAsJsonArray("body_route");
        List<Vec3> points = new ArrayList<>();
        int start = -1;
        for (int index = 0; index < route.size(); index++) {
            JsonArray node = route.get(index).getAsJsonArray();
            Vec3 feet = courtFeet(node, fixture);
            require(courtStanding(fixture.level, feet), "Courtyard body or floor obstructed " + node);
            points.add(feet);
            if (node.equals(fixture.spec.getAsJsonArray("entrance"))) start = index;
        }
        require(start >= 0, "Entry missing from declared body route");
        Set<Integer> visited = new HashSet<>();
        ArrayDeque<Integer> queue = new ArrayDeque<>();
        visited.add(start);
        queue.add(start);
        while (!queue.isEmpty()) {
            int current = queue.remove();
            JsonArray left = route.get(current).getAsJsonArray();
            for (int next = 0; next < points.size(); next++) {
                if (visited.contains(next)) continue;
                JsonArray right = route.get(next).getAsJsonArray();
                int distance = Math.abs(left.get(0).getAsInt() - right.get(0).getAsInt()) + Math.abs(left.get(2).getAsInt() - right.get(2).getAsInt());
                if (distance == 1 && courtEdge(fixture.level, points.get(current), points.get(next))) { visited.add(next); queue.add(next); }
            }
        }
        require(visited.size() == points.size(), "Courtyard upper/stair targets unreachable " + visited.size() + "/" + points.size());
        JsonArray sky = fixture.spec.getAsJsonArray("sky_sample");
        for (int vertical = sky.get(1).getAsInt(); vertical < fixture.size.getY(); vertical++) require(fixture.level.getBlockState(world(new BlockPos(sky.get(0).getAsInt(), vertical, sky.get(2).getAsInt()), fixture)).isAir(), "True sky sample blocked");
        int topAir = 0;
        for (int across = 0; across < 35; across++) for (int forward = 0; forward < 35; forward++) {
            BlockState top = fixture.level.getBlockState(world(new BlockPos(across,21,forward),fixture));
            boolean intended = across >= 8 && across <= 26 && forward >= 8 && forward <= 26;
            require(top.isAir() == intended, "Top skylight whitelist mismatch");
            if(top.isAir()) topAir++;
        }
        require(topAir == 361, "Skylight count mismatch");
        Map<String,Object> report = new LinkedHashMap<>();
        report.put("status","PASS");
        report.put("source_nbt_sha256",fixture.hash);
        report.put("body_targets",points.size());
        report.put("reached_targets",visited.size());
        report.put("upper_targets",fixture.spec.get("upper_count").getAsInt());
        report.put("stair_targets",fixture.spec.get("stair_count").getAsInt());
        report.put("body","0.6x1.8 native AABB, support slice and raised edge sweeps; max half-step0.5; no jump/fly shortcut");
        report.put("true_sky_sample",sky);
        report.put("top_air_whitelist",topAir);
        report.put("other_roof_columns",864);
        report.put("source_modified_for_analysis",false);
        try { Files.writeString(fixture.root.resolve("courtyard-analysis-" + fixture.rotation.name().toLowerCase() + ".json"),new GsonBuilder().setPrettyPrinting().create().toJson(report)+"\n"); }
        catch(Exception exception) { throw new IllegalStateException(exception); }
    }
